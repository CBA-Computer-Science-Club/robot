cmake_minimum_required(VERSION 3.8)
project(motor_controller)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_path(LGPIO_INCLUDE_DIR lgpio.h)
find_library(LGPIO_LIB lgpio)
if(NOT LGPIO_LIB OR NOT LGPIO_INCLUDE_DIR)
  message(FATAL_ERROR "lgpio development library is required for Raspberry Pi motor hardware")
endif()
add_executable(motor_controller_node src/motor_controller.cpp)
target_compile_features(motor_controller_node PRIVATE cxx_std_17)
target_include_directories(motor_controller_node PRIVATE ${LGPIO_INCLUDE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/include)
ament_target_dependencies(motor_controller_node rclcpp geometry_msgs nav_msgs sensor_msgs std_msgs)
target_link_libraries(motor_controller_node ${LGPIO_LIB})
install(TARGETS motor_controller_node DESTINATION lib/${PROJECT_NAME})
ament_package()
