cmake_minimum_required(VERSION 3.16)
project(memory_store_standalone LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
add_executable(memory_store_test memory_store_test.cpp ../src/memory_store.cpp)
target_include_directories(memory_store_test PRIVATE ../src ${NLOHMANN_JSON_INCLUDE_DIR})
enable_testing()
add_test(NAME memory_store_test COMMAND memory_store_test)
set_tests_properties(memory_store_test PROPERTIES TIMEOUT 10)
